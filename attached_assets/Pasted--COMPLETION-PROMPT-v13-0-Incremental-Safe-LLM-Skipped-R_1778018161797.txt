## ⚠️ IMPORTANT: LLM PLACEHOLDER RULE

Har text generation function (script, hook, caption, translation) mein:

```typescript
import { LlmGenerator } from './LlmGenerator';
import { RuleBasedGenerator } from './RuleBasedGenerator';

async function generateText(type: string, prompt: string): Promise<string> {
  try {
    // Primary: Try LLM if configured (abhi placeholder)
    if (LlmGenerator.isAvailable()) {
      return await LlmGenerator.generate(prompt);
    }
  } catch (e) {
    logger.warn('LLM failed, falling back to rule-based');
  }
  // Secondary: Rule-based deterministic
  return RuleBasedGenerator.generate(type, prompt);
}
